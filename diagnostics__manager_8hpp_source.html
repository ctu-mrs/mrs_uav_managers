349template <typename sh_T>
350DiagnosticsManager::subscriptionResult_t<sh_T> DiagnosticsManager::processIncomingMessage(mrs_lib::SubscriberHandler<sh_T> &sh) {
351 DiagnosticsManager::subscriptionResult_t<sh_T> msg;
352 msg.hasNewMessage = sh.newMsg();
353 msg.message = msg.hasNewMessage ? sh.getMsg() : sh.peekMsg();
354 if (msg.message != nullptr) {
355 if (clock_->now() - sh.lastMsgTime() > not_reporting_timeout_) {
356 msg.message = nullptr;
357 msg.hasNewMessage = true;
358 }
359 }
360 return msg;
361}

373template <typename sh_T>
374typename sh_T::ConstSharedPtr DiagnosticsManager::peekFreshMsg(const mrs_lib::SubscriberHandler<sh_T> &sh) const {
375 if (!sh.hasMsg() || clock_->now() - sh.lastMsgTime() > not_reporting_timeout_)
376 return nullptr;
377 return sh.peekMsg();
378}
